cmake_minimum_required(VERSION 3.18)
project(GTA6HUD CXX)
set(CMAKE_CXX_STANDARD 17)
# AML_DIR = hasil clone https://github.com/RusJJ/AndroidModLoader (berisi folder mod/)
set(AML_DIR "${CMAKE_SOURCE_DIR}/AndroidModLoader" CACHE PATH "Lokasi AndroidModLoader")
add_library(GTA6HUD SHARED src/main.cpp src/config.cpp)
target_include_directories(GTA6HUD PRIVATE ${AML_DIR})
target_compile_options(GTA6HUD PRIVATE -O2 -fvisibility=hidden -Wall -Wno-unused-function)
target_link_libraries(GTA6HUD PRIVATE log dl)
